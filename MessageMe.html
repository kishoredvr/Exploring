import { serve } from "https://deno.land/std@0.224.0/http/server.ts"; const FUNCTION_VERSION = "submit-form-v3-2026-09-20"; const SUPABASE_URL = (Deno.env.get("SUPABASE_URL") || Deno.env.get("CLBASE_URL") || "").replace(/\/+$/, ""); const CLBASE_KEY = Deno.env.get("CLBASE_KEY") || ""; const CLBASE_SERVICE_KEY = Deno.env.get("CLBASE_SERVICE_KEY") || Deno.env.get("SUPABASE_SERVICE_ROLE_KEY") || ""; const DB_KEY = CLBASE_SERVICE_KEY || CLBASE_KEY; const ORDERS_TABLE = Deno.env.get("PURGE_TABLE") || "orders"; const BREVO_KEY = Deno.env.get("BREVO_KEY") || ""; const MAIL_FROM = Deno.env.get("MAIL_FROM") || ""; const MAIL_FROM_NAME = Deno.env.get("MAIL_FROM_NAME") || "Parishodana"; const MAIL_TO = Deno.env.get("MAIL_TO") || ""; const ALLOWED_ORIGINS = new Set([ "https://parishodana.xyz", "https://kishoredvr.github.io", ]); function corsHeaders(req: Request) { const origin = req.headers.get("origin") || ""; return { "Access-Control-Allow-Origin": ALLOWED_ORIGINS.has(origin) ? origin : "*", "Access-Control-Allow-Methods": "POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type, Authorization, apikey, x-client-info", "Access-Control-Max-Age": "3600", "Vary": "Origin", "Content-Type": "application/json", }; } function jsonResponse(req: Request, status: number, body: Record) { return new Response(JSON.stringify(body), { status, headers: corsHeaders(req), }); } function clean(value: unknown): string { return value === undefined || value === null ? "" : String(value).trim(); } function escapeHtml(value: unknown): string { return clean(value) .replaceAll("&", "&") .replaceAll("<", "<") .replaceAll(">", ">") .replaceAll('"', """) .replaceAll("'", "'"); } /* Helper specifically implemented to convert newlines to
for cleanly rendering research payloads */ function formatMultiline(value: unknown): string { return escapeHtml(value).replace(/\n/g, "
"); } function validateConfig() { if (!SUPABASE_URL) throw new Error("SUPABASE_URL/CLBASE_URL is not configured"); if (!DB_KEY) throw new Error("CLBASE_KEY is not configured"); if (!BREVO_KEY) throw new Error("BREVO_KEY is not configured"); if (!MAIL_FROM) throw new Error("MAIL_FROM is not configured"); if (!MAIL_TO) throw new Error("MAIL_TO is not configured"); } function buildEmailHtml(orderData: Record, internalOrderId: string) { const purpose = clean(orderData.purpose); return ` New Form Submission

New Form Submission

${purpose === "Financial Services" ? `` : ""} ${purpose === "Software Development" ? `` : ""}
Purpose${escapeHtml(purpose)}
Name${escapeHtml(orderData.name)}
Email${escapeHtml(orderData.email)}
Mobile${escapeHtml(orderData.mobileNumber)}
Financial Service${escapeHtml(orderData.financial_service)}
Software Type${escapeHtml(orderData.software_type)}
Message / Expectations${formatMultiline(orderData.expectations)}
Submitted At${escapeHtml(orderData.submittedAt)}

Internal Order ID: ${escapeHtml(internalOrderId)}

Function Version: ${FUNCTION_VERSION}

`; } async function saveSubmission(orderData: Record, internalOrderId: string) { validateConfig(); const purpose = clean(orderData.purpose); // STRICT WHITELIST: only these columns are sent at the orders top level. const databaseRow = { merchantorderid: internalOrderId, amount: 0, status: "pending", mobile: clean(orderData.mobileNumber) || null, meta: { source: "submit-form", purpose, name: clean(orderData.name), email: clean(orderData.email), financial_service: clean(orderData.financial_service), software_type: clean(orderData.software_type), submittedAt: clean(orderData.submittedAt) || new Date().toISOString(), }, raw_payload: { ...orderData, purpose, service_area: purpose, savedBy: "submit-form", functionVersion: FUNCTION_VERSION, savedAt: new Date().toISOString(), }, submission_type: "form", }; console.log("FUNCTION VERSION:", FUNCTION_VERSION); console.log("DATABASE INSERT: USING STRICT WHITELIST"); console.log("TOP LEVEL DATABASE KEYS:", Object.keys(databaseRow)); // sb_secret_* is not a JWT. Use it as the PostgREST apikey and do not // send Authorization: Bearer . const response = await fetch(`${SUPABASE_URL}/rest/v1/${ORDERS_TABLE}`, { method: "POST", headers: { "Content-Type": "application/json", apikey: DB_KEY, ...(CLBASE_SERVICE_KEY && !CLBASE_SERVICE_KEY.startsWith("sb_secret_") ? { Authorization: `Bearer ${CLBASE_SERVICE_KEY}` } : {}), Prefer: "return=representation", }, body: JSON.stringify(databaseRow), }); const text = await response.text(); if (!response.ok) { console.error("DATABASE INSERT FAILED:", response.status, text); throw new Error(`Database insert failed (${response.status}): ${text}`); } const saved = JSON.parse(text); console.log("DATABASE INSERT SUCCESS:", saved); return saved; } async function sendBrevoEmail(orderData: Record, internalOrderId: string) { const purpose = clean(orderData.purpose); const response = await fetch("https://api.brevo.com/v3/smtp/email", { method: "POST", headers: { "Content-Type": "application/json", "api-key": BREVO_KEY, Accept: "application/json", }, body: JSON.stringify({ sender: { email: MAIL_FROM, name: MAIL_FROM_NAME }, to: [{ email: MAIL_TO }], subject: `New ${purpose || "Form"} Submission`, htmlContent: buildEmailHtml(orderData, internalOrderId), }), }); const text = await response.text(); if (!response.ok) { console.error("BREVO EMAIL FAILED:", response.status, text); throw new Error(`Brevo email failed (${response.status}): ${text}`); } let result: unknown = text; try { result = JSON.parse(text); } catch { /* keep text */ } console.log("BREVO EMAIL SUCCESS:", result); return result; } serve(async (req) => { if (req.method === "OPTIONS") return new Response(null, { status: 204, headers: corsHeaders(req) }); if (req.method !== "POST") return jsonResponse(req, 405, { success: false, error: "Method Not Allowed" }); try { const orderData = await req.json() as Record; const purpose = clean(orderData.purpose || orderData.service_area); const name = clean(orderData.name); const email = clean(orderData.email); const mobileNumber = clean(orderData.mobileNumber); if (!purpose) throw new Error("purpose is required"); if (!name) throw new Error("name is required"); if (!email) throw new Error("email is required"); if (!mobileNumber) throw new Error("mobileNumber is required"); orderData.purpose = purpose; orderData.service_area = purpose; orderData.name = name; orderData.email = email; orderData.mobileNumber = mobileNumber; orderData.submittedAt = clean(orderData.submittedAt) || new Date().toISOString(); const internalOrderId = `FORM-${Date.now()}-${crypto.randomUUID().slice(0, 8)}`; const databaseResult = await saveSubmission(orderData, internalOrderId); const emailResult = await sendBrevoEmail(orderData, internalOrderId); console.log("======================================"); console.log("SUBMIT FORM COMPLETE"); console.log(JSON.stringify({ purpose, internalOrderId, databaseSaved: true, emailSent: true }, null, 2)); console.log("======================================"); return jsonResponse(req, 200, { success: true, message: "Form submitted successfully", internalOrderId, databaseSaved: true, emailSent: true, databaseResult, emailResult, }); } catch (err) { console.error("SUBMIT FORM ERROR:", err); return jsonResponse(req, 500, { success: false, error: err instanceof Error ? err.message : String(err), functionVersion: FUNCTION_VERSION, }); } });
